######## SPOTLESS-GRADLE HOOK START ########

# NOTE: This hook will run spotlessApply on ALL kotlin files in the project, not only files in the staging area.

CHANGED_FILES="$(git --no-pager diff --name-status --no-color --cached | awk '$1 != "D" && $NF ~ /\.kts?$/ { print $NF }')"

if [ -n "$CHANGED_FILES" ]; then
    # A file that is staged AND modified in the working tree cannot be handled
    # safely: spotless rewrites the staged region, so the saved diff of the
    # unstaged changes no longer applies. Refuse before touching anything.
    mixed_files="$(
        {
            printf '%s\n' "$CHANGED_FILES" | sort -u
            git --no-pager diff --name-only --no-color | sort -u
        } | sort | uniq -d
    )"
    if [ -n "$mixed_files" ]; then
        echo "[error] These files have both staged and unstaged changes:"
        printf '%s\n' "$mixed_files" | sed 's/^/[error]   /'
        echo "[error] spotless cannot format them without risking your unstaged work."
        echo "[error] Stage or stash the remaining changes, then commit again."
        unset mixed_files
        exit 1
    fi
    unset mixed_files

    # Resolve the git dir instead of assuming ".git" is a directory: in a
    # worktree it is a file, and any path below it fails with ENOTDIR.
    git_dir="$(git rev-parse --absolute-git-dir)"
    diff="$git_dir/unstaged-spotless-git-hook.diff"

    if ! git diff --color=never > "$diff"; then
        echo "[error] Failed to save unstaged changes to $diff. Aborting commit."
        unset diff git_dir
        exit 1
    fi

    if [ -s "$diff" ]; then
      if ! git apply -R "$diff"; then
        echo "[error] Failed to set aside unstaged changes. Aborting commit."
        echo "[error] Your unstaged work is saved in: $diff"
        unset diff git_dir
        exit 1
      fi
    fi

    echo "Running spotlessApply"

    ./gradlew --quiet spotlessApply
    gradle_command_exit_code=$?

    echo "spotlessApply finished"

    echo "$CHANGED_FILES" | while read -r file; do
        if [ -f "$file" ]; then
            git add "$file"
        fi
    done

    if [ -s "$diff" ]; then
      if ! git apply --ignore-whitespace "$diff"; then
        echo "[error] Failed to restore unstaged changes. Aborting commit."
        echo "[error] This happens when spotless reformats a region the saved"
        echo "[error] diff quotes, e.g. after a commit made with --no-verify."
        echo "[error] Your unstaged work is saved in: $diff"
        echo "[error] Restore it manually with: git apply --3way $diff"
        unset diff git_dir
        exit 1
      fi
    fi
    rm -f "$diff"
    unset diff git_dir

    echo "Completed spotless hook."
    if [ $gradle_command_exit_code -ne 0 ]; then
        exit $gradle_command_exit_code
    fi
fi
######## SPOTLESS-GRADLE HOOK END ########
